Usage Examples

This page provides practical examples demonstrating various use cases of the PID Object Tracker library.

Example 1: Basic Tracking (Manual Updates)​

This example shows basic object tracking with manual detection updates, suitable for integration with custom detection pipelines.

#include "lib-pid-otrack.h"

using namespace DctotrackPID;

int main() {
    // Initialize tracker with model and display resolutions
    FrameResolution model_res(640, 384);
    FrameResolution display_res(1920, 1080);
    ObjectTrackerLib tracker(model_res, display_res);

    // Configure tracker
    tracker.initializeWithConfig("my_app");
    tracker.ai_obj_trk_set_target_id(1);
    tracker.ai_obj_trk_set_mode(TrackingMode::CENTROID);
    tracker.ai_obj_trk_enable();

    // Main tracking loop
    while (running) {
        // Get detections from your detection pipeline
        std::vector<DetectedObject> detections = getDetectionsFromPipeline();

        // Update tracker with new detections
        tracker.ai_obj_trk_update_detections(detections);

        // Calculate pan/tilt command
        PanTiltCommand cmd = tracker.calculatePanTiltCommand();

        if (cmd.valid) {
            printf("Pan: %.2f, Tilt: %.2f\n", cmd.pan_delta, cmd.tilt_delta);
            // Send commands to PTZ controller
            sendToPTZController(cmd.pan_delta, cmd.tilt_delta);
        }

        usleep(33000);  // ~30 Hz control rate
    }

    return 0;
}

Example 2: MQTT-Based Tracking​

This example demonstrates automatic tracking with MQTT-based detection updates, ideal for distributed systems.

#include "mqtt-object-tracker.h"

using namespace DctotrackPID;

int main() {
    // Initialize MQTT tracker
    FrameResolution model_res(640, 384);
    FrameResolution display_res(1920, 1080);
    MQTTObjectTracker mqtt_tracker(model_res, display_res);

    // Configure MQTT connection
    MQTTConfig mqtt_config;
    mqtt_config.broker_host = "192.168.1.100";
    mqtt_config.broker_port = 1883;
    mqtt_config.client_id = "camera_tracker";
    mqtt_config.detection_topic = "ai/detections/camera1";
    mqtt_config.username = "";  // Optional
    mqtt_config.password = "";  // Optional

    if (!mqtt_tracker.initializeMQTT(mqtt_config)) {
        fprintf(stderr, "Failed to initialize MQTT\n");
        return -1;
    }

    if (!mqtt_tracker.startMQTT()) {
        fprintf(stderr, "Failed to start MQTT\n");
        return -1;
    }

    // Configure tracking parameters
    auto* tracker = mqtt_tracker.getTracker();
    tracker->initializeWithConfig("mqtt_app");
    tracker->ai_obj_trk_set_target_id(1);
    tracker->ai_obj_trk_enable();

    // Main tracking loop
    // Detections are automatically received via MQTT
    while (running) {
        if (tracker->ai_obj_trk_is_enabled()) {
            PanTiltCommand cmd = tracker->calculatePanTiltCommand();

            if (cmd.valid) {
                // Control camera based on cmd.pan_delta and cmd.tilt_delta
                control_camera(cmd.pan_delta, cmd.tilt_delta);
            }
        }

        usleep(100000);  // 10 Hz control rate
    }

    // Cleanup
    mqtt_tracker.stopMQTT();
    return 0;
}

Example 3: ROI-Based Tracking​

This example shows how to use Region of Interest (ROI) tracking mode, where the camera only moves when the object approaches ROI boundaries.

#include "lib-pid-otrack.h"

using namespace DctotrackPID;

int main() {
    // Initialize tracker
    ObjectTrackerLib tracker(
        FrameResolution(640, 384),
        FrameResolution(1920, 1080)
    );

    tracker.initializeWithConfig("roi_app");

    // Configure ROI mode with custom ROI
    tracker.ai_obj_trk_set_mode(TrackingMode::ROI);

    // Define ROI as center 60% of frame
    float roi_x = 1920 * 0.2;       // 20% from left
    float roi_y = 1080 * 0.2;       // 20% from top
    float roi_width = 1920 * 0.6;   // 60% of width
    float roi_height = 1080 * 0.6;  // 60% of height

    tracker.ai_obj_trk_set_roi(roi_x, roi_y, roi_width, roi_height);

    // Select target and enable tracking
    tracker.ai_obj_trk_set_target_id(1);
    tracker.ai_obj_trk_enable();

    // Main tracking loop
    while (running) {
        std::vector<DetectedObject> detections = getDetections();
        tracker.ai_obj_trk_update_detections(detections);

        PanTiltCommand cmd = tracker.calculatePanTiltCommand();

        if (cmd.valid) {
            // Camera only moves when object approaches ROI edges
            controlPTZ(cmd.pan_delta, cmd.tilt_delta);
        }

        usleep(33000);
    }

    return 0;
}

Example 4: PID Tuning for Different Scenarios​

This example demonstrates how to tune PID parameters for different tracking scenarios.

#include "lib-pid-otrack.h"

using namespace DctotrackPID;

// Aggressive tracking (fast response, may be jittery)
void configureAggressiveTracking(ObjectTrackerLib& tracker) {
    tracker.ai_obj_trk_set_pid_gains(
        0.20f,  // pan_kp - higher for faster response
        0.002f, // pan_ki
        0.020f, // pan_kd
        0.20f,  // tilt_kp
        0.002f, // tilt_ki
        0.020f  // tilt_kd
    );

    tracker.ai_obj_trk_set_pid_limits(-80.0f, 80.0f);
    tracker.ai_obj_trk_set_filtering_params(0.02f, 0.6f);  // Less smoothing
}

// Smooth tracking (slower response, more stable)
void configureSmoothTracking(ObjectTrackerLib& tracker) {
    tracker.ai_obj_trk_set_pid_gains(
        0.08f,  // pan_kp - lower for gentler response
        0.001f, // pan_ki
        0.012f, // pan_kd - higher to reduce overshoot
        0.08f,  // tilt_kp
        0.001f, // tilt_ki
        0.012f  // tilt_kd
    );

    tracker.ai_obj_trk_set_pid_limits(-50.0f, 50.0f);
    tracker.ai_obj_trk_set_filtering_params(0.05f, 0.9f);  // More smoothing
}

// Balanced tracking (default settings)
void configureBalancedTracking(ObjectTrackerLib& tracker) {
    tracker.ai_obj_trk_set_pid_gains(
        0.10f,  // pan_kp
        0.001f, // pan_ki
        0.015f, // pan_kd
        0.10f,  // tilt_kp
        0.001f, // tilt_ki
        0.015f  // tilt_kd
    );

    tracker.ai_obj_trk_set_pid_limits(-60.0f, 60.0f);
    tracker.ai_obj_trk_set_filtering_params(0.03f, 0.8f);
}

int main() {
    ObjectTrackerLib tracker(
        FrameResolution(640, 384),
        FrameResolution(1920, 1080)
    );

    tracker.initializeWithConfig("tuning_app");

    // Choose tracking style based on your needs
    configureSmoothTracking(tracker);  // For surveillance cameras
    // configureAggressiveTracking(tracker);  // For sports tracking
    // configureBalancedTracking(tracker);    // For general use

    tracker.ai_obj_trk_set_target_id(1);
    tracker.ai_obj_trk_enable();

    // Tracking loop...

    return 0;
}

Example 5: Dynamic Topic Switching​

This example shows how to dynamically switch between different MQTT topics at runtime, useful for multi-camera systems.

#include "mqtt-object-tracker.h"

using namespace DctotrackPID;

void switchToCamera(MQTTObjectTracker& tracker,
                    const std::string& old_topic,
                    const std::string& new_topic) {
    // Unsubscribe from old camera topic
    if (!old_topic.empty()) {
        tracker.unsubscribe(old_topic);
        printf("Unsubscribed from: %s\n", old_topic.c_str());
    }

    // Subscribe to new camera topic
    if (tracker.subscribe(new_topic, 1)) {
        tracker.setDetectionTopic(new_topic);
        printf("Switched to: %s\n", new_topic.c_str());
    } else {
        fprintf(stderr, "Failed to subscribe to: %s\n", new_topic.c_str());
    }
}

int main() {
    MQTTObjectTracker tracker(
        FrameResolution(640, 384),
        FrameResolution(1920, 1080)
    );

    // Initialize MQTT
    MQTTConfig config;
    config.broker_host = "localhost";
    config.broker_port = 1883;
    config.client_id = "multi_camera_tracker";
    config.detection_topic = "camera/cam1/detections";

    tracker.initializeMQTT(config);
    tracker.startMQTT();

    auto* obj_tracker = tracker.getTracker();
    obj_tracker->initializeWithConfig("multi_cam_app");
    obj_tracker->ai_obj_trk_enable();

    std::string current_topic = "camera/cam1/detections";

    // Main loop
    while (running) {
        // Check for camera switch command (e.g., from user input)
        if (shouldSwitchCamera()) {
            std::string new_topic = getNewCameraTopic();
            switchToCamera(tracker, current_topic, new_topic);
            current_topic = new_topic;
        }

        // Process tracking
        PanTiltCommand cmd = obj_tracker->calculatePanTiltCommand();
        if (cmd.valid) {
            controlCamera(cmd.pan_delta, cmd.tilt_delta);
        }

        usleep(100000);
    }

    tracker.stopMQTT();
    return 0;
}

Example 6: Interactive Target Selection​

This example demonstrates interactive target selection by clicking on display coordinates.

#include "lib-pid-otrack.h"
#include <iostream>

using namespace DctotrackPID;

// Simulate user clicking at display coordinates
void handleUserClick(ObjectTrackerLib& tracker, float click_x, float click_y) {
    printf("User clicked at (%.1f, %.1f)\n", click_x, click_y);

    if (tracker.ai_obj_trk_set_by_coordinate(click_x, click_y)) {
        int selected_id = tracker.ai_obj_trk_get_target_id();
        printf("Selected object ID: %d\n", selected_id);

        DetectedObject target;
        if (tracker.ai_obj_trk_get_current_target(target)) {
            printf("Target at (%.1f, %.1f) with confidence %.2f\n",
                   target.x, target.y, target.confidence);
        }
    } else {
        printf("No object found near click position\n");
    }
}

// Display available objects
void showAvailableObjects(ObjectTrackerLib& tracker) {
    auto ids = tracker.ai_obj_trk_get_available_ids();

    printf("Available object IDs: ");
    for (int id : ids) {
        printf("%d ", id);
    }
    printf("\n");
}

int main() {
    ObjectTrackerLib tracker(
        FrameResolution(640, 384),
        FrameResolution(1920, 1080)
    );

    tracker.initializeWithConfig("interactive_app");
    tracker.ai_obj_trk_set_mode(TrackingMode::CENTROID);
    tracker.ai_obj_trk_enable();

    while (running) {
        // Get detections
        std::vector<DetectedObject> detections = getDetections();
        tracker.ai_obj_trk_update_detections(detections);

        // Show available objects
        showAvailableObjects(tracker);

        // Simulate user interaction
        // In a real app, this would come from mouse clicks or touch events
        if (userClickedScreen()) {
            float click_x, click_y;
            getUserClickCoordinates(click_x, click_y);
            handleUserClick(tracker, click_x, click_y);
        }

        // Calculate and apply tracking commands
        PanTiltCommand cmd = tracker.calculatePanTiltCommand();
        if (cmd.valid) {
            controlPTZ(cmd.pan_delta, cmd.tilt_delta);
        }

        usleep(33000);
    }

    return 0;
}

Example 7: Multi-Resolution Support​

This example shows how to dynamically update resolutions when switching between different video sources.

#include "lib-pid-otrack.h"

using namespace DctotrackPID;

struct CameraConfig {
    int model_width, model_height;
    int display_width, display_height;
    std::string name;
};

void switchCamera(ObjectTrackerLib& tracker, const CameraConfig& config) {
    printf("Switching to camera: %s\n", config.name.c_str());
    printf("Model res: %dx%d, Display res: %dx%d\n",
           config.model_width, config.model_height,
           config.display_width, config.display_height);

    // Update resolutions
    tracker.ai_obj_trk_set_resolutions(
        config.model_width, config.model_height,
        config.display_width, config.display_height
    );

    // Reset PID controllers when switching cameras
    tracker.ai_obj_trk_reset_pid_controllers();
}

int main() {
    // Define camera configurations
    CameraConfig cam1 = {640, 384, 1920, 1080, "HD Camera 1"};
    CameraConfig cam2 = {320, 192, 1280, 720, "SD Camera 2"};
    CameraConfig cam3 = {640, 384, 3840, 2160, "4K Camera 3"};

    // Initialize with first camera
    ObjectTrackerLib tracker(
        FrameResolution(cam1.model_width, cam1.model_height),
        FrameResolution(cam1.display_width, cam1.display_height)
    );

    tracker.initializeWithConfig("multi_res_app");
    tracker.ai_obj_trk_enable();

    CameraConfig* current_cam = &cam1;

    while (running) {
        // Check for camera switch
        if (shouldSwitchCamera()) {
            CameraConfig* new_cam = selectNewCamera();
            switchCamera(tracker, *new_cam);
            current_cam = new_cam;
        }

        // Normal tracking operation
        std::vector<DetectedObject> detections = getDetections();
        tracker.ai_obj_trk_update_detections(detections);

        PanTiltCommand cmd = tracker.calculatePanTiltCommand();
        if (cmd.valid) {
            controlPTZ(cmd.pan_delta, cmd.tilt_delta);
        }

        usleep(33000);
    }

    return 0;
}

Common Patterns​

Error Handling​

// Check if target exists before tracking
DetectedObject target;
if (tracker.ai_obj_trk_get_current_target(target)) {
    // Target is valid, proceed with tracking
    PanTiltCommand cmd = tracker.calculatePanTiltCommand();
    // ...
} else {
    // No target found or tracking disabled
    printf("No active target\n");
}

Debug Logging​

// Enable debug logging for troubleshooting
ObjectTrackerLib::setLogLevel(LogLevel::LOG_DEBUG);

// Disable logging in production
ObjectTrackerLib::setLogLevel(LogLevel::NONE);

Status Monitoring​

// Print tracker status
tracker.printStatus();

// Check MQTT status
if (mqtt_tracker.isMQTTConnected()) {
    mqtt_tracker.printMQTTStatus();
}

Integration with GStreamer​

Example showing integration with GStreamer-based detection pipeline:

// Callback from GStreamer detection element
void on_detection_callback(std::vector<DetectedObject> detections, void* user_data) {
    ObjectTrackerLib* tracker = static_cast<ObjectTrackerLib*>(user_data);

    // Update tracker with new detections
    tracker->ai_obj_trk_update_detections(detections);

    // Calculate pan/tilt command
    PanTiltCommand cmd = tracker->calculatePanTiltCommand();

    if (cmd.valid) {
        // Send to PTZ controller
        send_ptz_command(cmd.pan_delta, cmd.tilt_delta);
    }
}
