API Reference

The dctotrack plugin provides a comprehensive C API for programmatic control and configuration. This allows applications to configure and manage the tracking element without using GStreamer property strings.

Initialization​

api_dctotrack_init​

Initialize and create a dctotrack element within a pipeline.

Signature:

GstElement* api_dctotrack_init(GstElement *pipeline, const char *element_name);

Parameters:

  • pipeline: Pointer to the GStreamer pipeline
  • element_name: Name for the dctotrack element (can be NULL for auto-naming)

Returns:

  • GstElement*: Pointer to the created dctotrack element, or NULL on failure

Example:

GstElement *pipeline = gst_pipeline_new("tracking-pipeline");
GstElement *dctotrack = api_dctotrack_init(pipeline, "tracker");

if (dctotrack == NULL) {
    g_printerr("Failed to create dctotrack element\n");
    return -1;
}

Tracking Parameter Setters​

api_dctotrack_set_track_thresh​

Set the minimum confidence threshold for starting a new track.

Signature:

bool api_dctotrack_set_track_thresh(GstElement *element, float thresh);

Parameters:

  • element: dctotrack element pointer
  • thresh: Track threshold (0.0 - 1.0)

Returns:

  • bool: true on success, false on failure

Example:

if (!api_dctotrack_set_track_thresh(dctotrack, 0.6)) {
    g_printerr("Failed to set track threshold\n");
}

api_dctotrack_set_high_thresh​

Set the high confidence threshold for first-stage association.

Signature:

bool api_dctotrack_set_high_thresh(GstElement *element, float thresh);

Parameters:

  • element: dctotrack element pointer
  • thresh: High threshold (0.0 - 1.0)

Returns:

  • bool: true on success, false on failure

Example:

api_dctotrack_set_high_thresh(dctotrack, 0.7);

api_dctotrack_set_match_thresh​

Signature:

bool api_dctotrack_set_match_thresh(GstElement *element, float thresh);

Parameters:

  • element: dctotrack element pointer
  • thresh: Match threshold (0.0 - 1.0)

Returns:

  • bool: true on success, false on failure

Example:

api_dctotrack_set_match_thresh(dctotrack, 0.8);

api_dctotrack_set_track_buffer​

Set the number of frames to keep lost tracks before removal.

Signature:

bool api_dctotrack_set_track_buffer(GstElement *element, int buffer);

Parameters:

  • element: dctotrack element pointer
  • buffer: Track buffer size (1 - 1000 frames)

Returns:

  • bool: true on success, false on failure

Example:

api_dctotrack_set_track_buffer(dctotrack, 50);

api_dctotrack_set_frame_rate​

Set the expected frame rate for motion prediction.

Signature:

bool api_dctotrack_set_frame_rate(GstElement *element, int fps);

Parameters:

Returns:

Complete Application Example​

Basic Tracking Application​

#include <gst/gst.h>
#include "apidctotrack.h"

int main(int argc, char *argv[]) {
    GstElement *pipeline, *source, *convert1, *dctobd, *dctotrack, *convert2, *sink;
    GstBus *bus;
    GstMessage *msg;

    // Initialize GStreamer
    gst_init(&argc, &argv);

    // Create pipeline
    pipeline = gst_pipeline_new("tracking-pipeline");

    // Create elements
    source = gst_element_factory_make("v4l2src", "source");
    convert1 = gst_element_factory_make("videoconvert", "convert1");
    dctobd = gst_element_factory_make("dctobd", "detector");
    dctotrack = api_dctotrack_init(pipeline, "tracker");
    convert2 = gst_element_factory_make("videoconvert", "convert2");
    sink = gst_element_factory_make("autovideosink", "sink");

    if (!pipeline || !source || !convert1 || !dctobd || !dctotrack || !convert2 || !sink) {
        g_printerr("Not all elements could be created\n");
        return -1;
    }

    // Configure DCTOBD
    g_object_set(dctobd, "model", "/path/to/model.synap", NULL);
    g_object_set(dctobd, "threshold", 0.4, NULL);

    // Configure dctotrack using API
    api_dctotrack_set_track_thresh(dctotrack, 0.5);
    api_dctotrack_set_high_thresh(dctotrack, 0.6);
    api_dctotrack_set_match_thresh(dctotrack, 0.8);
    api_dctotrack_set_track_buffer(dctotrack, 30);
    api_dctotrack_set_frame_rate(dctotrack, 30);
    api_dctotrack_set_output_mode(dctotrack, "both");
    api_dctotrack_set_output_format(dctotrack, "tracking");
    api_dctotrack_set_mqtt_broker(dctotrack, "localhost");
    api_dctotrack_set_mqtt_port(dctotrack, 1883);
    api_dctotrack_set_mqtt_topic(dctotrack, "tracking/camera1");
    api_dctotrack_set_track_statistics(dctotrack, true);

    // Add elements to pipeline
    gst_bin_add_many(GST_BIN(pipeline), source, convert1, dctobd, dctotrack,
                     convert2, sink, NULL);

    // Link elements
    if (!gst_element_link_many(source, convert1, dctobd, dctotrack,
                               convert2, sink, NULL)) {
        g_printerr("Elements could not be linked\n");
        gst_object_unref(pipeline);
        return -1;
    }

    // Print configuration
    float track_thresh, high_thresh, match_thresh;
    int track_buffer, frame_rate, mqtt_port;
    char output_mode[64], output_format[64], mqtt_broker[256], mqtt_topic[256];
    bool stats_enabled;

    api_dctotrack_get_track_thresh(dctotrack, &track_thresh);
    api_dctotrack_get_high_thresh(dctotrack, &high_thresh);
    api_dctotrack_get_match_thresh(dctotrack, &match_thresh);
    api_dctotrack_get_track_buffer(dctotrack, &track_buffer);
    api_dctotrack_get_frame_rate(dctotrack, &frame_rate);
    api_dctotrack_get_output_mode(dctotrack, output_mode, sizeof(output_mode));
    api_dctotrack_get_output_format(dctotrack, output_format, sizeof(output_format));
    api_dctotrack_get_mqtt_broker(dctotrack, mqtt_broker, sizeof(mqtt_broker));
    api_dctotrack_get_mqtt_port(dctotrack, &mqtt_port);
    api_dctotrack_get_mqtt_topic(dctotrack, mqtt_topic, sizeof(mqtt_topic));
    api_dctotrack_get_track_statistics(dctotrack, &stats_enabled);

    g_print("\n=== DCToTrack Configuration ===\n");
    g_print("Track Threshold: %.2f\n", track_thresh);
    g_print("High Threshold: %.2f\n", high_thresh);
    g_print("Match Threshold: %.2f\n", match_thresh);
    g_print("Track Buffer: %d frames\n", track_buffer);
    g_print("Frame Rate: %d fps\n", frame_rate);
    g_print("Output Mode: %s\n", output_mode);
    g_print("Output Format: %s\n", output_format);
    g_print("MQTT Broker: %s:%d\n", mqtt_broker, mqtt_port);
    g_print("MQTT Topic: %s\n", mqtt_topic);
    g_print("Statistics: %s\n", stats_enabled ? "enabled" : "disabled");
    g_print("===============================\n\n");

    // Start playing
    gst_element_set_state(pipeline, GST_STATE_PLAYING);

    // Wait for error or EOS
    bus = gst_element_get_bus(pipeline);
    msg = gst_bus_timed_pop_filtered(bus, GST_CLOCK_TIME_NONE,
                                      GST_MESSAGE_ERROR | GST_MESSAGE_EOS);

    // Parse message
    if (msg != NULL) {
        GError *err;
        gchar *debug_info;

        switch (GST_MESSAGE_TYPE(msg)) {
            case GST_MESSAGE_ERROR:
                gst_message_parse_error(msg, &err, &debug_info);
                g_printerr("Error: %s\n", err->message);
                g_error_free(err);
                g_free(debug_info);
                break;
            case GST_MESSAGE_EOS:
                g_print("End-Of-Stream reached\n");
                break;
            default:
                break;
        }
        gst_message_unref(msg);
    }

    // Clean up
    gst_object_unref(bus);
    gst_element_set_state(pipeline, GST_STATE_NULL);
    gst_object_unref(pipeline);

    return 0;
}

Compilation​

gcc -o tracking_app tracking_app.c \
    $(pkg-config --cflags --libs gstreamer-1.0) \
    -lapidctotrack \
    -L/usr/local/lib

Error Handling​

All setter and getter functions return bool:

  • true: Success
  • false: Failure (invalid element, invalid parameter, etc.)

Always check return values:

if (!api_dctotrack_set_track_thresh(dctotrack, 0.6)) {
    g_printerr("Failed to set track threshold - element may be NULL or value out of range\n");
    // Handle error appropriately
}
